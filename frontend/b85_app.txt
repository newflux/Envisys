function createRadialAlpha(size: number, fadeStart: number, fadeEnd: number) {
  const canvas = document.createElement('canvas')
  canvas.width = size; canvas.height = size
  const ctx = canvas.getContext('2d')
  if (!ctx) return null
  const grd = ctx.createRadialGradient(size/2, size/2, fadeStart, size/2, size/2, fadeEnd)
  grd.addColorStop(0, 'white')
  grd.addColorStop(1, 'black')
  ctx.fillStyle = grd
  ctx.fillRect(0, 0, size, size)
  return new CanvasTexture(canvas)
}

function createEmissiveGradient() {
  const canvas = document.createElement('canvas')
  canvas.width = 256

}

const addFresnelRim = (mat: any, baseStrength: number) => {
  mat.customProgramCacheKey = () => 'fresnelRim'
  mat.onBeforeCompile = (shader: any) => {
    shader.uniforms.uReveal = { value: 0.0 }
    shader.uniforms.uBaseStr = { value: baseStrength }
    mat.userData.shader = shader
    shader.vertexShader = shader.vertexShader.replace(
      '#include <common>',
      `#include <common>
       varying vec3 vViewPositionFresnel;
       varying vec3 vNormalFresnel;`
    )
    shader.vertexShader = shader.vertexShader.replace(
      '#include <worldpos_vertex>',
      `#include <worldpos_vertex>
       vViewPositionFresnel = -mvPosition.xyz;
       vNormalFresnel = normalize(normalMatrix * normal);`
    )
    shader.fragmentShader = shader.fragmentShader.replace(
      '#include <common>',
      `#include <common>
       uniform float uReveal;
       uniform float uBaseStr;
       varying vec3 vViewPositionFresnel;
       varying vec3 vNormalFresnel;`
    )
    shader.fragmentShader = shader.fragmentShader.replace(
      '#include <dithering_fragment>',
      `#include <dithering_fragment>
       vec3 normalF = normalize(vNormalFresnel);
       vec3 viewDirF = normalize(vViewPositionFresnel);
       float rimDot = max(dot(normalF, viewDirF), 0.0);
       float rim = pow(1.0 - rimDot, 3.0);
       float rimStrength = mix(0.25, 0.9, uReveal) * uBaseStr;
       vec3 rimColor = vec3(0.81, 0.88, 1.0);
       gl_FragColor.rgb += rimColor * rim * rimStrength;`
    )
  }
}

      mouseOffsetY += (targetMouseOffsetY - mouseOffsetY) * 0.05

      // Idle Drift
      const driftX = Math.sin(t * 0.5) * 0.03
      const driftY = Math.cos(t * 0.4) * 0.03

      // Yaw rotation (parallax up to ±4°)
      let targetYaw = targetMouseX * MathUtils.degToRad(4)
      currentYaw += (targetYaw - currentYaw) * 0.05
      if (chamberScene.value) {
         chamberScene.value.rotation.y = currentYaw
      }

      const currentDist = baseDistance - (scrollProxy.camZ * (baseDistance * 0.4))
      const currentTargetY = baseCameraTarget.y + (scrollProxy.camZ * (baseSizeY * 0.1))
      const currentCamY = baseCameraPos.y + (scrollProxy.camZ * (baseSizeY * 0.2))

      cameraRef.value.position.set(
        baseCameraPos.x + mouseOffsetX + driftX,
        currentCamY + mouseOffsetY + driftY,
        baseCameraTarget.z + currentDist
      )
      
      cameraRef.value.lookAt(baseCameraTarget.x, currentTargetY, baseCameraTarget.z)
    }
  })
  gsap.ticker.lagSmoothing(0)

  // 2. Custom Cursor & Interactive Background Glow
  const cursor = document.querySelector('.cursor') as HTMLElement;
  const layoutPage = document.querySelector('.layout__page') as HTMLElement;
  
  if (cursor) {
    cursor.style.display = 'block';
    window.addEventListener('mousemove', (e) => {
      // Move cursor (centered on mouse)
      gsap.to(cursor, {
        x: e.clientX,
        y: e.clientY,
        xPercent: -50,
        yPercent: -50,
        duration: 0.1,
        ease: 'power2.out'
      })
      
      // Move CSS Gradient center
      const mouseXPct = (e.clientX / window.innerWidth) * 100;
      const mouseYPct = (e.clientY / window.innerHeight) * 100;
      if (layoutPage) {
        gsap.to(layoutPage, {
          '--mouse-x': `${mouseXPct}%`,
          '--mouse-y': `${mouseYPct}%`,
          duration: 1,
          ease: 'power2.out'
        });
      }
      
      // Update target for the render loop to interpolate
      targetMouseX = (e.clientX / window.innerWidth) * 2 - 1;
    });
  }

  // 3. Load both GLBs directly with GLTFLoader for material access + animation
  const dracoLoader = new DRACOLoader()
  dracoLoader.setDecoderPath('https://www.gstatic.com/draco/versioned/decoders/1.5.7/')
  const gltfLoader = new GLTFLoader()
  gltfLoader.setDRACOLoader(dracoLoader)

  // Load CHAMBER — upgrade materials + hook up AnimationMixer
  gltfLoader.load('/models/chamber_main.glb', (gltf) => {
    // Precise material mapping based on the provided GLTF node names
    gltf.scene.traverse((child: any) => {
      if (child.isMesh && child.material) {
        child.material.fog = false // Only the hall is fogged
        
        const mat = child.material
        const name: string = mat.name || ''
        
        // ** THE BLENDER NORMALS FIX **
        mat.side = 2 // THREE.DoubleSide
        
        if (name === 'Cavity_Exterior.001') {
          // MeshStandardMaterial body
          const stdMat = new MeshStandardMaterial({
            color: '#030303',
            roughness: 0.15,
            metalness: 0.9,
            envMapIntensity: 0.4
          })
          addFresnelRim(stdMat, 1.0)
          child.material = stdMat
          bodyMat = stdMat
        } else if (name === 'Black') {
          // Silver center column
          mat.color?.set('#111111')
          mat.metalness = 0.9
          mat.roughness = 0.2
          mat.envMapIntensity = 0.4
          addFresnelRim(mat, 0.4)
          silverMat = mat
        } else if (name === 'Cavity_Interior.001') {
          // Emissive interior walls
          mat.color?.set('#1a0500')
          mat.emissive?.set('#ffffff')
          if (windowEmissiveTex) mat.emissiveMap = windowEmissiveTex
          mat.emissiveIntensity = 0.6
          mat.toneMapped = false
          interiorMat = mat
        } else if (name.toLowerCase().includes('logo')) {
          mat.color?.set('#e6efe4')
          // ensure no emissive blue
          mat.emissive?.set('#000000')
        } else if (mat.emissive && (mat.emissive.r > 0 || mat.emissive.g > 0 || mat.emissive.b > 0)) {
          mat.toneMapped = false
        } else {
          // Fallback
          mat.metalness = 0.5
          mat.roughness = 0.5
        }
      }
    })

    // Store and hook up AnimationMixer to this exact scene
    chamberScene.value = gltf.scene
    
    // Duplicate Coplanar Detector
    gltf.scene.updateMatrixWorld(true)
    const meshes: any[] = []
    
    gltf.scene.traverse((child: any) => {
      if (child.isMesh && child.material) {
        child.geometry.computeBoundingBox()
        const box = child.geometry.boundingBox.clone()
        box.applyMatrix4(child.matrixWorld)
        
        meshes.push({
          child,
          box,
          name: child.name,
          matName: child.material.name,
          color: child.material.color ? child.material.color.getHexString() : 'N/A'
        })
      }
    })

    console.log('[DEBUG] Meshes parsed:', meshes)

    for (let i = 0; i < meshes.length; i++) {
      for (let j = i + 1; j < meshes.length; j++) {
        const b1 = meshes[i].box
        const b2 = meshes[j].box
        if (b1.min.distanceTo(b2.min) < 0.001 && b1.max.distanceTo(b2.max) < 0.001) {
          console.warn(`[DUPLICATE DETECTED] ${meshes[i].name} overlaps exactly with ${meshes[j].name}. Hiding duplicate.`)
          meshes[j].child.visible = false
        }
      }
    }
    
    // Compute exact bounding box post-scale
    const box = new Box3().setFromObject(gltf.scene)
    const size = new Vector3()
    box.getSize(size)
    size.multiplyScalar(2.5) // The template applies 2.5 scale
    baseSizeY = size.y
    
    const center = new Vector3()
    box.getCenter(center)
    center.multiplyScalar(2.5)
    center.add(new Vector3(0, -2.05, 0)) // The template position
    
    baseCameraTarget.copy(center)
    
    // Compute distance for 45% frame height
    const fov = 30
    baseDistance = (size.y / 0.45) / (2 * Math.tan(MathUtils.degToRad(fov / 2)))
    
    // Lazy-load warehouse
    requestAnimationFrame(() => {
      requestAnimationFrame(() => {
        const startTime = performance.now()
        gltfLoader.load('/models/3D_Warehouse_2_opt.glb', (gltf) => {
          const loadTime = performance.now() - startTime
          console.log(`[Perf] Warehouse Environment loaded in ${loadTime.toFixed(0)} ms`)
          
          const wBox = new Box3().setFromObject(gltf.scene)
          const wSize = new Vector3()
          wBox.getSize(wSize)
          
          // Meters per unit
          const metersPerUnit = 0.3048
          const hallScale = baseSizeY / (2.2 / metersPerUnit)
          
          gltf.scene.scale.setScalar(hallScale)
          
          // Floor logic
          // The actual floor is the bottom of the bounding box.
          const chamberFloorY = -2.05
          const scaledMinY = wBox.min.y * hallScale
          const offsetY = chamberFloorY - scaledMinY
          
          // We'll calculate offsetZ via placement grid search
          let offsetZ = 20
          let offsetX = 0
          
          const raycaster = new THREE.Raycaster()
          const chamberW = size.x
          const chamberD = size.z
          const candidates = []
          
          const maxHallHeight = wSize.y * hallScale
          
          for (let hz = -60; hz <= 60; hz += 4) {
            for (let hx = -20; hx <= 20; hx += 4) {
              gltf.scene.position.set(hx, offsetY, hz)
              gltf.scene.updateMatrixWorld(true)
              
              let valid = true
              const points = [
                new Vector3(0, chamberFloorY + baseSizeY, 0),
                new Vector3(-chamberW/2, chamberFloorY + baseSizeY, -chamberD/2),
                new Vector3(chamberW/2, chamberFloorY + baseSizeY, -chamberD/2),
                new Vector3(-chamberW/2, chamberFloorY + baseSizeY, chamberD/2),
                new Vector3(chamberW/2, chamberFloorY + baseSizeY, chamberD/2)
              ]
              
              for (let pt of points) {
                raycaster.set(pt, new Vector3(0, 1, 0))
                const hits = raycaster.intersectObject(gltf.scene, true)
                if (hits.length > 0) {
                  const hit = hits[0]
                  if (hit.distance < 0.8 * maxHallHeight) {
                    valid = false
                    break
                  }
                }
              }
              
              if (valid) {
                raycaster.set(new Vector3(0, chamberFloorY + 1, 0), new Vector3(0, 0, -1))
                const hits = raycaster.intersectObject(gltf.scene, true)
                let clearDist = 1000
                if (hits.length > 0) clearDist = hits[0].distance
                if (clearDist >= 40) {
                  // Reward positive hz because that pushes hall back, making lamp rows visible behind
                  candidates.push({ hx, hz, clearDist })
                }
              }
            }
          }
          
          candidates.sort((a, b) => b.hz - a.hz) // prefer maximum hz (lamp rows visible behind)
          console.log('[Placement] Top 5 candidates:', candidates.slice(0, 5))
          
          if (candidates.length > 0) {
            const best = candidates[0]
            offsetX = best.hx
            offsetZ = best.hz
          }
          
          gltf.scene.position.set(offsetX, offsetY, offsetZ)
          
          // Print raycast hit for center
          raycaster.set(new Vector3(0, chamberFloorY + baseSizeY, 0), new Vector3(0, 1, 0))
          const centerHits = raycaster.intersectObject(gltf.scene, true)
          if (centerHits.length > 0) {
            console.log(`[Diagnose Placement] Upward ray from center hit: ${centerHits[0].object.name} at distance ${centerHits[0].distance.toFixed(2)}`)
          }
          
          // Z of front wall
          const scaledMaxZ = wBox.max.z * hallScale
          const frontWallZ = scaledMaxZ + offsetZ
          const depth = wSize.z * hallScale
          const frontMargin = depth * 0.1
          console.log(`[Placement] Hall Scale: ${hallScale.toFixed(3)}, OffsetY: ${offsetY.toFixed(3)}, OffsetZ: ${offsetZ.toFixed(3)}`)
          console.log(`[Placement] Chamber stands at Z=0. Hall rear is at ${(-40).toFixed(1)}, front is at ${frontWallZ.toFixed(1)}. Camera is at ${baseDistance.toFixed(1)}.`)
          console.log(`[Placement] Margin from camera to front wall: ${(frontWallZ - baseDistance).toFixed(1)} (Required: ${frontMargin.toFixed(1)})`)
          
          // Extract materials and hide glass
          let meshCount = 0
          gltf.scene.traverse((child: any) => {
            if (child.isMesh && child.material) {
              meshCount++
              const mat = child.material
              const matName = mat.name.toLowerCase()
              
              if (matName.includes('glass')) {
                child.visible = false
              } else if (mat.name.includes('827713812')) {
                // Bulb material
                mat.emissive.setHex(0xffaa55) // warm-white
                mat.toneMapped = false
                mat.emissiveIntensity = 0
                bulbMats.push(mat)
              } else {
                if (!warehouseMatBaseColor.has(mat)) {
                  warehouseMatBaseColor.set(mat, mat.color.clone())
                  warehouseMatNames.add(mat.name)
                }
              }
            }
          })
          
          console.log('[Darkness] Applied lerp to materials:', Array.from(warehouseMatNames))
          
          warehouseScene.value = gltf.scene
          
          // Compute Fog densities
          const fogTargetDist = baseDistance + (60 * hallScale)
          const density0 = 1.7308 / fogTargetDist
          const density1 = 0.957 / fogTargetDist
          console.log(`[Fog] Distance ${fogTargetDist.toFixed(1)} -> Density0: ${density0.toFixed(4)}, Density1: ${density1.toFixed(4)}`)
          
          // Inject fog update directly into ticker
          gsap.ticker.add(() => {
            if (fogRef.value) {
              const r = isDebug ? (debugReveal !== null ? debugReveal : reveal.value) : scrollProxy.reveal
              fogRef.value.density = density0 + (density1 - density0) * r
            }
          })
          
          console.log(`[Warehouse] Meshes (Draw Calls): ${meshCount}`)
        })
      })
    })
    
    // Lower camera base height to drop the horizon behind the base
    // "Camera height is the floor top plus 0.3 x chamber height, looking at the chamber's center"
    const chamberWorldHeight = baseSizeY
    const chamberFloorY = -2.05
    baseCameraPos.set(center.x, chamberFloorY + (0.3 * chamberWorldHeight), center.z + baseDistance)
    baseCameraTarget.copy(center)
    
    if (cameraRef.value) {
      cameraRef.value.position.copy(baseCameraPos)
      cameraRef.value.lookAt(baseCameraTarget)
    }

    if (gltf.animations && gltf.animations.length > 0) {
      doorAnim.mixer = new AnimationMixer(gltf.scene)
      const action = doorAnim.mixer.clipAction(gltf.animations[0])
      action.setLoop(THREE.LoopOnce, 1)
      action.clampWhenFinished = true
      action.paused = true
      action.timeScale = 0
      action.play()
      doorAnim.action = action
      doorAnim.duration = gltf.animations[0].duration
      console.log('[Chamber] Ready. Animation:', gltf.animations[0].name, '| duration:', doorAnim.duration)
    } else {
      console.warn('[Chamber] No animations found in chamber_main.glb')
    }
  })

  // 4. ScrollTrigger for Camera Z-Axis (Zoom + door animation)
  setTimeout(() => {
    if (cameraRef.value) {
      // Master GSAP ScrollTrigger
      const tl = gsap.timeline({
        scrollTrigger: {
          trigger: '.layout__page',
          scroller: '#scroller',
          start: 'top top',
          end: 'bottom bottom',
          scrub: 1,
          onUpdate: () => {
            const p = tl.progress()
            const crosses = [0, 0.20, 0.40, 0.60]
            crosses.forEach(cp => {
              if ((lastScrollP <= cp && p >= cp) || (lastScrollP >= cp && p <= cp)) {
                console.log(`[VERIFY] Scroll crossed ${cp*100}% | door: ${scrollProxy.door.toFixed(3)} | action time: ${(doorAnim.action ? doorAnim.action.time : 0).toFixed(3)}`)
              }
            })
            lastScrollP = p
          }
        }
      })

      // Reveal 0 to 1 over first 35%
      tl.to(scrollProxy, { reveal: 1, ease: 'none', duration: 0.35 }, 0)
      // Door & CamZ from 15% to 60%
      tl.to(scrollProxy, { door: 1, camZ: 1, ease: 'power2.inOut', duration: 0.45 }, 0.15)
      
      // Ensure the timeline spans the full scroll distance (duration 1)
      tl.to({}, { duration: 1 }, 0)
    }
  }, 500)
})
</script>
